    68→}
    69→
    70→const STATUS_META: Record<string, { label: string; cls: string }> = {
    71→  idea: { label: 'Ideia', cls: 'bg-muted text-muted-foreground' },
    72→  planned: { label: 'Planejada', cls: 'bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300' },
    73→  used: { label: 'Usada', cls: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300' },
    74→}
    75→
    76→export function IdeasSection() {
    77→  const companyId = useAppStore((s) => s.selectedCompanyId)
    78→  const setSection = useAppStore((s) => s.setSection)
    79→  const setSelectedCompany = useAppStore((s) => s.setSelectedCompany)
    80→  const setCreatorPrefill = useAppStore((s) => s.setCreatorPrefill)
    81→
    82→  const [genOpen, setGenOpen] = useState(false)
    83→  const [genNiche, setGenNiche] = useState('')
    84→  const [genPlatforms, setGenPlatforms] = useState<string[]>(['instagram', 'facebook'])
    85→  const [genCount, setGenCount] = useState(8)
    86→  const [generating, setGenerating] = useState(false)
    87→  const [search, setSearch] = useState('')
    88→  const [filterCat, setFilterCat] = useState('all')
    89→  const [filterStatus, setFilterStatus] = useState('all')
    90→  const [deleteId, setDeleteId] = useState<string | null>(null)
    91→  const [addOpen, setAddOpen] = useState(false)
    92→  const [manualIdea, setManualIdea] = useState({ title: '', description: '', category: 'educacional', platform: 'instagram' })
    93→
    94→  const url = companyId ? `/api/ideas?companyId=${companyId}` : '/api/ideas'
    95→  const { data: ideasData, loading, refresh } = useFetch<any>(url, [companyId])
    96→  const { data: compData } = useFetch<any>('/api/companies', [])
    97→
    98→  const ideas = ideasData?.ideas || []
    99→  const companies = compData?.companies || []
   100→  const activeCompany = companies.find((c: any) => c.id === companyId)
   101→
   102→  const filtered = ideas.filter((idea: any) => {
   103→    if (search && !idea.title.toLowerCase().includes(search.toLowerCase()) && !idea.description.toLowerCase().includes(search.toLowerCase())) return false
   104→    if (filterCat !== 'all' && idea.category !== filterCat) return false
   105→    if (filterStatus !== 'all' && idea.status !== filterStatus) return false
   106→    return true
   107→  })
   108→
   109→  // Stats
   110→  const byStatus = {
   111→    idea: ideas.filter((i: any) => i.status === 'idea').length,
   112→    planned: ideas.filter((i: any) => i.status === 'planned').length,
   113→    used: ideas.filter((i: any) => i.status === 'used').length,
   114→  }
   115→  const avgScore = ideas.length
   116→    ? Math.round(ideas.reduce((a: number, i: any) => a + (i.score || 0), 0) / ideas.length)
   117→    : 0
   118→
   119→  async function handleGenerate() {
   120→    if (!companyId) {
   121→      toast.error('Selecione uma empresa no topo da tela primeiro')
   122→      return
   123→    }
   124→    if (!genNiche.trim() && !activeCompany?.niche) {
   125→      toast.error('Informe o nicho da empresa')
   126→      return
   127→    }
   128→    setGenerating(true)
   129→    try {
   130→      const res = await apiPost<any>('/api/ideas/generate', {
   131→        company: activeCompany?.name || 'Empresa',
   132→        niche: genNiche.trim() || activeCompany?.niche || 'geral',
   133→        platforms: genPlatforms,
   134→        count: genCount,
   135→        companyId,
   136→        save: true,
   137→      })
   138→      toast.success(`${res.count} ideias geradas! 💡`)
   139→      setGenOpen(false)
   140→      setGenNiche('')
   141→      refresh()
   142→    } catch (e: any) {
   143→      toast.error(e.message || 'Falha ao gerar ideias')
   144→    } finally {
   145→      setGenerating(false)
   146→    }
   147→  }
   148→
   149→  async function toggleStatus(idea: any) {
   150→    const next = idea.status === 'idea' ? 'planned' : idea.status === 'planned' ? 'used' : 'idea'
   151→    try {
   152→      await apiPost(`/api/ideas?id=${idea.id}`, { status: next })
   153→      refresh()
   154→    } catch {
   155→      toast.error('Erro ao atualizar')
   156→    }
   157→  }
   158→
   159→  async function handleDelete(id: string) {
   160→    try {
   161→      await apiDelete(`/api/ideas?id=${id}`)
   162→      toast.success('Ideia excluída')
   163→      refresh()
   164→    } catch {
   165→      toast.error('Erro ao excluir')
   166→    } finally {
   167→      setDeleteId(null)
   168→    }
   169→  }
   170→
   171→  async function handleAddManual() {
   172→    if (!manualIdea.title.trim() || !manualIdea.description.trim()) {
   173→      toast.error('Preencha título e descrição')
   174→      return
   175→    }
   176→    try {
   177→      await apiPost('/api/ideas', { ...manualIdea, companyId })
   178→      toast.success('Ideia adicionada')
   179→      setAddOpen(false)
   180→      setManualIdea({ title: '', description: '', category: 'educacional', platform: 'instagram' })
   181→      refresh()
   182→    } catch {
   183→      toast.error('Erro ao adicionar')
   184→    }
   185→  }
   186→
   187→  function togglePlatform(p: string) {
   188→    setGenPlatforms((arr) => (arr.includes(p) ? arr.filter((x) => x !== p) : [...arr, p]))
   189→  }
   190→
   191→  return (
   192→    <div className="space-y-6">
   193→      <SectionHeader
   194→        title="Banco de Ideias"
   195→        description="Nunca mais fique sem ideias. Gere um calendário de conteúdo com IA."
   196→        icon={Lightbulb}
   197→        action={
   198→          <div className="flex gap-2">
   199→            <Button variant="outline" className="gap-2" onClick={() => setAddOpen(true)}>
   200→              <Plus className="w-4 h-4" />
   201→              <span className="hidden sm:inline">Ideia manual</span>
   202→            </Button>
   203→            <Button
   204→              className="gap-2 bg-gradient-to-r from-primary to-fuchsia-500 hover:opacity-90"
   205→              onClick={() => setGenOpen(true)}
   206→            >
   207→              <Sparkles className="w-4 h-4" />
   208→              Gerar com IA
   209→            </Button>
   210→          </div>
   211→        }
   212→      />
   213→
   214→      {/* Stats */}
   215→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   216→        <Card className="p-4 flex items-center gap-3">
   217→          <div className="w-10 h-10 rounded-lg bg-primary/10 text-primary flex items-center justify-center">
   218→            <Lightbulb className="w-5 h-5" />
   219→          </div>
   220→          <div>
   221→            <p className="text-xl font-bold tabular-nums">{ideas.length}</p>
   222→            <p className="text-xs text-muted-foreground">Total de ideias</p>
   223→          </div>
   224→        </Card>
   225→        <Card className="p-4 flex items-center gap-3">
   226→          <div className="w-10 h-10 rounded-lg bg-amber-500/10 text-amber-500 flex items-center justify-center">
   227→            <Clock className="w-5 h-5" />
   228→          </div>
   229→          <div>
   230→            <p className="text-xl font-bold tabular-nums">{byStatus.idea}</p>
   231→            <p className="text-xs text-muted-foreground">A explorar</p>
   232→          </div>
   233→        </Card>
   234→        <Card className="p-4 flex items-center gap-3">
   235→          <div className="w-10 h-10 rounded-lg bg-emerald-500/10 text-emerald-500 flex items-center justify-center">
   236→            <Calendar className="w-5 h-5" />
   237→          </div>
   238→          <div>
   239→            <p className="text-xl font-bold tabular-nums">{byStatus.planned}</p>
   240→            <p className="text-xs text-muted-foreground">Planejadas</p>
   241→          </div>
   242→        </Card>
   243→        <Card className="p-4 flex items-center gap-3">
   244→          <div className="w-10 h-10 rounded-lg bg-fuchsia-500/10 text-fuchsia-500 flex items-center justify-center">
   245→            <TrendingUp className="w-5 h-5" />
   246→          </div>
   247→          <div>
   248→            <p className="text-xl font-bold tabular-nums">{avgScore}</p>
   249→            <p className="text-xs text-muted-foreground">Score médio</p>
   250→          </div>
   251→        </Card>
   252→      </div>
   253→
   254→      {/* Filters */}
   255→      <Card className="p-3">
   256→        <div className="flex flex-wrap items-center gap-2">
   257→          <div className="relative flex-1 min-w-[200px]">
   258→            <Input
   259→              placeholder="Buscar ideias..."
   260→              value={search}
   261→              onChange={(e) => setSearch(e.target.value)}
   262→              className="pl-8 h-9"
   263→            />
   264→            <Filter className="absolute left-2.5 top-1/2 -translate-y-1/2 w-3.5 h-3.5 text-muted-foreground" />
   265→          </div>
   266→          <Select value={filterCat} onValueChange={setFilterCat}>
   267→            <SelectTrigger className="w-[150px] h-9">
   268→              <SelectValue placeholder="Categoria" />
   269→            </SelectTrigger>
   270→            <SelectContent>
   271→              <SelectItem value="all">Todas categorias</SelectItem>
   272→              {Object.entries(CATEGORY_META).map(([k, v]) => (
   273→                <SelectItem key={k} value={k}>
   274→                  {v.label}
   275→                </SelectItem>
   276→              ))}
   277→            </SelectContent>
   278→          </Select>
   279→          <Select value={filterStatus} onValueChange={setFilterStatus}>
   280→            <SelectTrigger className="w-[130px] h-9">
   281→              <SelectValue placeholder="Status" />
   282→            </SelectTrigger>
   283→            <SelectContent>
   284→              <SelectItem value="all">Todos status</SelectItem>
   285→              <SelectItem value="idea">Ideia</SelectItem>
   286→              <SelectItem value="planned">Planejada</SelectItem>
   287→              <SelectItem value="used">Usada</SelectItem>
   288→            </SelectContent>
   289→          </Select>
   290→          {companyId && (
   291→            <Button variant="ghost" size="sm" className="h-9" onClick={() => setSelectedCompany(null)}>
   292→              Limpar filtro
   293→            </Button>
   294→          )}
   295→        </div>
   296→      </Card>
   297→
   298→      {/* Ideas list */}
   299→      {loading ? (
   300→        <div className="space-y-2">
   301→          {Array.from({ length: 5 }).map((_, i) => (
   302→            <Skeleton key={i} className="h-24 rounded-xl" />
   303→          ))}
   304→        </div>
   305→      ) : filtered.length === 0 ? (
   306→        <Card>
   307→          <CardContent className="py-12">
   308→            <EmptyState
   309→              icon={Lightbulb}
   310→              title={ideas.length === 0 ? 'Nenhuma ideia ainda' : 'Nenhuma ideia encontrada'}
   311→              description={
   312→                ideas.length === 0
   313→                  ? 'Gere um calendário de ideias de conteúdo com IA baseado no nicho da sua empresa, ou adicione ideias manualmente.'
   314→                  : 'Ajuste os filtros para ver mais ideias.'
   315→              }
   316→              action={
   317→                ideas.length === 0 ? (
   318→                  <Button
   319→                    className="gap-2 bg-gradient-to-r from-primary to-fuchsia-500 hover:opacity-90"
   320→                    onClick={() => setGenOpen(true)}
   321→                  >
   322→                    <Sparkles className="w-4 h-4" />
   323→                    Gerar ideias com IA
   324→                  </Button>
   325→                ) : undefined
   326→              }
   327→            />
   328→          </CardContent>
   329→        </Card>
   330→      ) : (
   331→        <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   332→          <AnimatePresence>
   333→            {filtered.map((idea: any, i: number) => {
   334→              const catMeta = CATEGORY_META[idea.category] || CATEGORY_META.educacional
   335→              const statusMeta = STATUS_META[idea.status] || STATUS_META.idea
   336→              return (
   337→                <motion.div
   338→                  key={idea.id}
   339→                  initial={{ opacity: 0, y: 8 }}
   340→                  animate={{ opacity: 1, y: 0 }}
   341→                  exit={{ opacity: 0, y: -8 }}
   342→                  transition={{ delay: Math.min(i * 0.03, 0.3) }}
   343→                  className="group relative rounded-xl border border-border bg-card p-4 hover:shadow-md hover:border-primary/40 transition-all"
   344→                >
   345→                  {/* Left color stripe */}
   346→                  <div
   347→                    className="absolute left-0 top-4 bottom-4 w-1 rounded-full"
   348→                    style={{ backgroundColor: catMeta.color }}
   349→                  />
   350→                  <div className="pl-2">
   351→                    <div className="flex items-start gap-2 mb-2">
   352→                      <div className="flex-1 min-w-0">
   353→                        <div className="flex items-center gap-2 flex-wrap mb-1">
   354→                          <span
   355→                            className="text-[10px] font-semibold px-1.5 py-0.5 rounded"
   356→                            style={{ backgroundColor: `${catMeta.color}20`, color: catMeta.color }}
   357→                          >
   358→                            {catMeta.label}
   359→                          </span>
   360→                          <span className={cn('text-[10px] font-semibold px-1.5 py-0.5 rounded', statusMeta.cls)}>
   361→                            {statusMeta.label}
   362→                          </span>
   363→                          {idea.score >= 80 && (
   364→                            <span className="inline-flex items-center gap-0.5 text-[10px] font-semibold px-1.5 py-0.5 rounded bg-rose-100 text-rose-700 dark:bg-rose-950 dark:text-rose-300">
   365→                              <Flame className="w-2.5 h-2.5" />
   366→                              Hot
   367→                            </span>
   368→                          )}
   369→                        </div>
   370→                        <h3 className="font-semibold text-sm leading-tight">{idea.title}</h3>
   371→                      </div>
   372→                      <div className="flex items-center gap-1 shrink-0">
   373→                        <div className="relative w-9 h-9">
   374→                          <svg className="w-9 h-9 -rotate-90" viewBox="0 0 36 36">
   375→                            <circle cx="18" cy="18" r="15" fill="none" stroke="var(--muted)" strokeWidth="3" />
   376→                            <circle
   377→                              cx="18"
   378→                              cy="18"
   379→                              r="15"
   380→                              fill="none"
   381→                              stroke={idea.score >= 80 ? '#F43F5E' : idea.score >= 60 ? '#10B981' : '#6B7280'}
   382→                              strokeWidth="3"
   383→                              strokeDasharray={`${(idea.score / 100) * 94.2} 94.2`}
   384→                              strokeLinecap="round"
   385→                            />
   386→                          </svg>
   387→                          <span className="absolute inset-0 flex items-center justify-center text-[10px] font-bold">
   388→                            {idea.score}
   389→                          </span>
   390→                        </div>
   391→                      </div>
   392→                    </div>
   393→
   394→                    <p className="text-xs text-muted-foreground line-clamp-2 mb-2.5">{idea.description}</p>
   395→
   396→                    {idea.angle && (
   397→                      <div className="text-[11px] text-muted-foreground italic mb-2 line-clamp-1">
   398→                        💡 {idea.angle}
   399→                      </div>
   400→                    )}
   401→
   402→                    <div className="flex items-center gap-2 flex-wrap text-[10px] text-muted-foreground mb-2.5">
   403→                      <span className="flex items-center gap-1">
   404→                        <PlatformBadge platform={idea.platform} />
   405→                        {idea.platform}
   406→                      </span>
   407→                      {idea.bestDay && (
   408→                        <span className="flex items-center gap-1">
   409→                          <Calendar className="w-3 h-3" />
   410→                          {idea.bestDay}
   411→                        </span>
   412→                      )}
   413→                      {idea.bestTime && (
   414→                        <span className="flex items-center gap-1">
   415→                          <Clock className="w-3 h-3" />
   416→                          {idea.bestTime}
   417→                        </span>
   418→                      )}
   419→                    </div>
   420→
   421→                    {idea.hashtags && (() => {
   422→                      let tags: string[] = []
   423→                      try {
   424→                        tags = JSON.parse(idea.hashtags)
   425→                      } catch {
   426→                        tags = []
   427→                      }
   428→                      return tags.length > 0 ? (
   429→                        <div className="flex flex-wrap gap-1 mb-2.5">
   430→                          {tags.slice(0, 4).map((t: string) => (
   431→                            <span key={t} className="text-[9px] px-1.5 py-0.5 rounded bg-primary/10 text-primary">
   432→                              #{t}
   433→                            </span>
   434→                          ))}
   435→                          {tags.length > 4 && (
   436→                            <span className="text-[9px] px-1.5 py-0.5 rounded bg-muted text-muted-foreground">
   437→                              +{tags.length - 4}
   438→                            </span>
   439→                          )}
   440→                        </div>
   441→                      ) : null
   442→                    })()}
   443→
   444→                    <div className="flex items-center gap-1.5 pt-1 border-t border-border">
   445→                      <Button
   446→                        size="sm"
   447→                        variant="ghost"
   448→                        className="h-7 text-xs gap-1 flex-1"
   449→                        onClick={() => toggleStatus(idea)}
   450→                      >
   451→                        {idea.status === 'idea' && <Calendar className="w-3 h-3" />}
   452→                        {idea.status === 'planned' && <Flame className="w-3 h-3" />}
   453→                        {idea.status === 'used' && <Lightbulb className="w-3 h-3" />}
   454→                        {idea.status === 'idea' ? 'Planejar' : idea.status === 'planned' ? 'Marcar usada' : 'Reabrir'}
   455→                      </Button>
   456→                      <Button
   457→                        size="sm"
   458→                        variant="ghost"
   459→                        className="h-7 text-xs gap-1"
   460→                        onClick={() => {
   461→                          if (idea.companyId) setSelectedCompany(idea.companyId)
   462→                          setCreatorPrefill({
   463→                            topic: `${idea.title}: ${idea.description}`,
   464→                            category: idea.category,
   465→                            platforms: [idea.platform],
   466→                            source: 'idea',
   467→                          })
   468→                          setSection('creator')
   469→                          toast.success('Ideia enviada para o criador')
   470→                        }}
   471→                      >
   472→                        <ArrowRight className="w-3 h-3" />
   473→                        Criar
   474→                      </Button>
   475→                      <Button
   476→                        size="icon"
   477→                        variant="ghost"
   478→                        className="w-7 h-7 opacity-0 group-hover:opacity-100 hover:text-rose-500 transition-opacity"
   479→                        onClick={() => setDeleteId(idea.id)}
   480→                      >
   481→                        <Trash2 className="w-3.5 h-3.5" />
   482→                      </Button>
   483→                    </div>
   484→                  </div>
   485→                </motion.div>
   486→              )
   487→            })}
   488→          </AnimatePresence>
   489→        </div>
   490→      )}
   491→
   492→      {/* Generate dialog */}
   493→      <Dialog open={genOpen} onOpenChange={setGenOpen}>
   494→        <DialogContent className="max-w-md">
   495→          <DialogHeader>
   496→            <DialogTitle className="flex items-center gap-2">
   497→              <div className="w-8 h-8 rounded-lg bg-gradient-to-br from-primary to-fuchsia-500 flex items-center justify-center">
   498→                <Wand2 className="w-4 h-4 text-white" />
   499→              </div>
   500→              Gerar ideias com IA
   501→            </DialogTitle>
