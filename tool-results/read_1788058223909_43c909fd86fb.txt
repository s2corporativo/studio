    71→function safeParse<T>(s: string | null | undefined, fallback: T): T {
    72→  if (!s) return fallback
    73→  try {
    74→    return JSON.parse(s) as T
    75→  } catch {
    76→    return fallback
    77→  }
    78→}

    80→function isSameDay(a: Date, b: Date) {
    81→  return (
    82→    a.getFullYear() === b.getFullYear() &&
    83→    a.getMonth() === b.getMonth() &&
    84→    a.getDate() === b.getDate()
    85→  )
    86→}

   556→      {/* Edit dialog */}

   572→  )

   578→  onSaved,
   579→  onDeleted,
   580→}: {
   581→  post: any
   582→  onClose: () => void
   583→  onSaved: () => void
   584→  onDeleted: () => void
   585→}) {
   586→  const [title, setTitle] = useState(post.title || '')
   587→  const [content, setContent] = useState(post.content || '')
   588→  const [scheduledAt, setScheduledAt] = useState(
   589→    toLocalInputValue(post.scheduledAt)
   590→  )
   591→  const [status, setStatus] = useState(post.status || 'draft')
   592→  const [category, setCategory] = useState(post.category || 'promocional')
   593→  const [tone, setTone] = useState(post.tone || 'profissional')
   594→  const [platforms, setPlatforms] = useState<string[]>(
   595→    (post.targets || []).map((t: any) => t.platform)
   596→  )
   597→  const [saving, setSaving] = useState(false)
   598→  const [deleting, setDeleting] = useState(false)
   599→
   600→  const hashtags = safeParse<string[]>(post.hashtags, [])
   601→
   602→  const togglePlatform = (p: string) => {
   603→    setPlatforms((prev) =>
   604→      prev.includes(p) ? prev.filter((x) => x !== p) : [...prev, p]
   605→    )
   606→  }
   607→
   608→  const handleSave = async () => {
   609→    if (!title.trim()) {
   610→      toast.error('Título é obrigatório')
   611→      return
   612→    }
   613→    setSaving(true)
   614→    try {
   615→      await apiPut(`/api/posts/${post.id}`, {
   616→        title,
   617→        content,
   618→        scheduledAt: scheduledAt ? new Date(scheduledAt).toISOString() : null,
   619→        status,
   620→        category,
   621→        tone,
   622→        platforms,
   623→        hashtags,
   624→        variations: {},
   625→      })
   626→      toast.success('Post atualizado com sucesso!')
   627→      onSaved()
   628→    } catch (e: any) {
   629→      toast.error(e?.message || 'Erro ao salvar post')
   630→    } finally {
   631→      setSaving(false)
   632→    }
   633→  }
   634→
   635→  const handleDelete = async () => {
   636→    setDeleting(true)
   637→    try {
   638→      await apiDelete(`/api/posts/${post.id}`)
   639→      toast.success('Post excluído')
   640→      onDeleted()
   641→    } catch (e: any) {
   642→      toast.error(e?.message || 'Erro ao excluir')
   643→    } finally {
   644→      setDeleting(false)
   645→    }
   646→  }
   647→
   648→  return (
   649→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   650→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   651→        <DialogHeader>
   652→          <DialogTitle className="flex items-center gap-2">
   653→            <Pencil className="w-5 h-5 text-primary" /> Editar Post
   654→          </DialogTitle>
   655→          <DialogDescription>
   656→            Altere os campos abaixo e salve as alterações.
   657→          </DialogDescription>
   658→        </DialogHeader>
   659→
   660→        <div className="space-y-4 py-2">
   661→          <div>
   662→            <Label>Título</Label>
   663→            <Input
   664→              value={title}
   665→              onChange={(e) => setTitle(e.target.value)}
   666→              className="mt-1.5"
   667→            />
   668→          </div>
   669→          <div>
   670→            <Label>Conteúdo</Label>
   671→            <Textarea
   672→              value={content}
   673→              onChange={(e) => setContent(e.target.value)}
   674→              rows={5}
   675→              className="mt-1.5 resize-none"
   676→            />
   677→            <p className="text-xs text-muted-foreground mt-1">
   678→              {content.length} caracteres
   679→            </p>
   680→          </div>
   681→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   682→            <div>
   683→              <Label>Agendar para</Label>
   684→              <Input
   685→                type="datetime-local"
   686→                value={scheduledAt}
   687→                onChange={(e) => setScheduledAt(e.target.value)}
   688→                className="mt-1.5"
   689→              />
   690→            </div>
   691→            <div>
   692→              <Label>Status</Label>
   693→              <Select value={status} onValueChange={setStatus}>
   694→                <SelectTrigger className="mt-1.5 w-full">
   695→                  <SelectValue />
   696→                </SelectTrigger>
   697→                <SelectContent>
   698→                  <SelectItem value="draft">Rascunho</SelectItem>
   699→                  <SelectItem value="scheduled">Agendado</SelectItem>
   700→                  <SelectItem value="published">Publicado</SelectItem>
   701→                </SelectContent>
   702→              </Select>
   703→            </div>
   704→          </div>
   705→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   706→            <div>
   707→              <Label>Categoria</Label>
   708→              <Select value={category} onValueChange={setCategory}>
   709→                <SelectTrigger className="mt-1.5 w-full">
   710→                  <SelectValue />
   711→                </SelectTrigger>
   712→                <SelectContent>
   713→                  {POST_CATEGORIES.map((c) => (
   714→                    <SelectItem key={c.value} value={c.value}>
   715→                      {c.label}
   716→                    </SelectItem>
   717→                  ))}
   718→                </SelectContent>
   719→              </Select>
   720→            </div>
   721→            <div>
   722→              <Label>Tom de voz</Label>
   723→              <Select value={tone} onValueChange={setTone}>
   724→                <SelectTrigger className="mt-1.5 w-full">
   725→                  <SelectValue />
   726→                </SelectTrigger>
   727→                <SelectContent>
   728→                  {TONES.map((t) => (
   729→                    <SelectItem key={t} value={t} className="capitalize">
   730→                      {t}
   731→                    </SelectItem>
   732→                  ))}
   733→                </SelectContent>
   734→              </Select>
   735→            </div>
   736→          </div>
   737→          <div>
   738→            <Label>Plataformas</Label>
   739→            <div className="grid grid-cols-2 sm:grid-cols-3 gap-2 mt-1.5">
   740→              {PLATFORMS.map((p) => {
   741→                const checked = platforms.includes(p)
   742→                const meta = PLATFORM_META[p as Platform]
   743→                return (
   744→                  <label
   745→                    key={p}
   746→                    className={cn(
   747→                      'flex items-center gap-2 p-2.5 rounded-lg border cursor-pointer transition-colors',
   748→                      checked
   749→                        ? 'border-primary bg-primary/5'
   750→                        : 'border-border hover:bg-accent/40'
   751→                    )}
   752→                  >
   753→                    <Checkbox
   754→                      checked={checked}
   755→                      onCheckedChange={() => togglePlatform(p)}
   756→                    />
   757→                    <PlatformIcon platform={p} className="w-4 h-4 shrink-0" />
   758→                    <span className="text-sm truncate">{meta.label}</span>
   759→                  </label>
   760→                )
   761→              })}
   762→            </div>
   763→          </div>
   764→          {hashtags.length > 0 && (
   765→            <div>
   766→              <Label>Hashtags</Label>
   767→              <div className="flex flex-wrap gap-1.5 mt-1.5">
   768→                {hashtags.map((h, i) => (
   769→                  <Badge key={i} variant="secondary" className="text-xs">
   770→                    #{h}
   771→                  </Badge>
   772→                ))}
   773→              </div>
   774→            </div>
   775→          )}
   776→        </div>
   777→
   778→        <DialogFooter className="flex items-center justify-between gap-2 sm:justify-between">
   779→          <Button
   780→            variant="destructive"
   781→            onClick={handleDelete}
   782→            disabled={deleting}
   783→            className="gap-2"
   784→          >
   785→            <Trash2 className="w-4 h-4" />
   786→            {deleting ? 'Excluindo...' : 'Excluir'}
   787→          </Button>
   788→          <div className="flex gap-2">
   789→            <DialogClose asChild>
   790→              <Button variant="outline">Cancelar</Button>
   791→            </DialogClose>
   792→            <Button onClick={handleSave} disabled={saving} className="gap-2">
   793→              {saving ? 'Salvando...' : 'Salvar'}
   794→            </Button>
   795→          </div>
   796→        </DialogFooter>
   797→      </DialogContent>
   798→    </Dialog>
   799→  )
   800→}
