    93→function safeParse<T>(s: string | null | undefined, fallback: T): T {
    94→  if (!s) return fallback
    95→  try {
    96→    return JSON.parse(s) as T
    97→  } catch {
    98→    return fallback
    99→  }
   100→}

   102→function isSameDay(a: Date, b: Date) {
   103→  return (
   104→    a.getFullYear() === b.getFullYear() &&
   105→    a.getMonth() === b.getMonth() &&
   106→    a.getDate() === b.getDate()
   107→  )
   108→}

  1002→function EditPostDialog({

  1005→  onSaved,
  1006→  onDeleted,

  1008→  post: any
  1009→  onClose: () => void
  1010→  onSaved: () => void
  1011→  onDeleted: () => void
  1012→}) {
  1013→  const [title, setTitle] = useState(post.title || '')
  1014→  const [content, setContent] = useState(post.content || '')
  1015→  const [scheduledAt, setScheduledAt] = useState(
  1016→    toLocalInputValue(post.scheduledAt)

  1022→    (post.targets || []).map((t: any) => t.platform)

  1028→
  1029→  const togglePlatform = (p: string) => {
  1030→    setPlatforms((prev) =>
  1031→      prev.includes(p) ? prev.filter((x) => x !== p) : [...prev, p]
  1032→    )
  1033→  }

  1037→      toast.error('Título é obrigatório')
  1038→      return
  1039→    }

  1045→        scheduledAt: scheduledAt ? new Date(scheduledAt).toISOString() : null,
  1046→        status,

  1060→  }

  1062→  const handleDelete = async () => {

  1065→      await apiDelete(`/api/posts/${post.id}`)

  1068→    } catch (e: any) {
  1069→      toast.error(e?.message || 'Erro ao excluir')

  1080→            <Pencil className="w-5 h-5 text-primary" /> Editar Post
  1081→          </DialogTitle>
  1082→          <DialogDescription>
  1083→            Altere os campos abaixo e salve as alterações.
  1084→          </DialogDescription>
  1085→        </DialogHeader>
  1086→
  1087→        <div className="space-y-4 py-2">
  1088→          <div>
  1089→            <Label>Título</Label>
  1090→            <Input
  1091→              value={title}
  1092→              onChange={(e) => setTitle(e.target.value)}
  1093→              className="mt-1.5"
  1094→            />
  1095→          </div>
  1096→          <div>
  1097→            <Label>Conteúdo</Label>
  1098→            <Textarea
  1099→              value={content}
  1100→              onChange={(e) => setContent(e.target.value)}
  1101→              rows={5}
  1102→              className="mt-1.5 resize-none"
  1103→            />
  1104→            <p className="text-xs text-muted-foreground mt-1">
  1105→              {content.length} caracteres
  1106→            </p>
  1107→          </div>
  1108→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
  1109→            <div>
  1110→              <Label>Agendar para</Label>
  1111→              <Input
  1112→                type="datetime-local"
  1113→                value={scheduledAt}
  1114→                onChange={(e) => setScheduledAt(e.target.value)}

  1119→              <Label>Status</Label>
  1120→              <Select value={status} onValueChange={setStatus}>
  1121→                <SelectTrigger className="mt-1.5 w-full">
  1122→                  <SelectValue />
  1123→                </SelectTrigger>
  1124→                <SelectContent>
